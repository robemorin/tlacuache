Ejemplo de Guía en Python usando SymPy y Matplotlib:

A continuación se muestra un código ejemplo que calcula el polinomio característico, valores y vectores propios de una matriz de $2 \times 2$, y posteriormente genera un gráfico que muestra un vector propio $v$ y su transformación lineal $Av$.

import sympy as sp import numpy as np import matplotlib.pyplot as plt # 1. Definir la matriz de manera simbólica A = sp.Matrix([[3, 2], [0, 1]]) # Definir la variable simbólica lambda lam = sp.symbols('lambda') # a) Obtener el polinomio característico p_char = A.charpoly(lam) print("--- Polinomio Característico ---") print(f"p(lambda) = {p_char.as_expr()}\n") # b) Obtener valores y vectores propios # eigenvects() devuelve una lista de tuplas: (valor_propio, multiplicidad, [vectores_propios]) par_propio = A.eigenvects() print("--- Valores y Vectores Propios ---") for val, mult, vects in par_propio: print(f"Valor propio: {val} (Multiplicidad: {mult})") for v in vects: print(f" Vector propio asociado:\n{v}\n") # c) Visualización gráfica (Ejemplo para el valor propio lambda = 3, v = [1, 0]) # Convertimos la matriz A a formato numérico de NumPy para operar con gráficos A_np = np.array(A.tolist(), dtype=float) # Vector propio asociado a lambda = 3 v = np.array([1, 0]) # Calcular la transformación del vector propio Av = A_np.dot(v) # Graficar plt.figure(figsize=(6, 6)) # Dibujar vector propio v plt.quiver(0, 0, v[0], v[1], angles='xy', scale_units='xy', scale=1, color='blue', label=f'v (Vector propio)') # Dibujar vector transformado Av plt.quiver(0, 0, Av[0], Av[1], angles='xy', scale_units='xy', scale=1, color='red', linestyle='--', alpha=0.7, label=f'A*v (Transformado, lambda*v)') plt.xlim(-5, 5) plt.ylim(-5, 5) plt.axhline(0, color='black', linewidth=0.8) plt.axvline(0, color='black', linewidth=0.8) plt.grid(True, which='both', linestyle=':', alpha=0.5) plt.legend() plt.title("Efecto de la transformación lineal en un Vector Propio") plt.show()
