Ejemplo de SVD y Medición de Tiempo para $A^n$ en Python:

El siguiente código ilustra cómo obtener la SVD usando NumPy y cómo medir la diferencia en tiempo de cómputo para la potencia de una matriz:

import numpy as np import time # --- Parte 1: Descomposición SVD --- A = np.array([[3.0, 0.0], [4.0, 5.0]]) # linalg.svd retorna U, la diagonal S (como vector de valores singulares) y V^T U, s, Vt = np.linalg.svd(A) S = np.diag(s) # Convertir el vector s a matriz diagonal print("--- Descomposición SVD de A ---") print("U =\n", U) print("S =\n", S) print("V^T =\n", Vt) print("Reconstrucción (U * S * V^T) =\n", U.dot(S).dot(Vt), "\n") # --- Parte 2: Eficiencia en Potencias de Matrices (C^n) --- C = np.array([[2.0, 1.0], [0.0, 3.0]]) n = 50000 # Método 1: Multiplicación sucesiva iterativa start_time = time.time() C_mult = np.eye(2) for _ in range(n): C_mult = C_mult.dot(C) time_mult = time.time() - start_time # Método 2: Usando Diagonalización C = P * D * P^-1 # Obtenemos valores y vectores propios de C eigenvalues, P = np.linalg.eig(C) P_inv = np.linalg.inv(P) start_time = time.time() D_n = np.diag(eigenvalues ** n) C_diag = P.dot(D_n).dot(P_inv) time_diag = time.time() - start_time print("--- Comparación de Tiempos para C^n (n = 50,000) ---") print(f"Método por multiplicación iterativa: {time_mult:.6f} segundos") print(f"Método por diagonalización: {time_diag:.6f} segundos") print(f"¿Los resultados coinciden?: {np.allclose(C_mult, C_diag)}") print(f"La diagonalización es {time_mult / max(time_diag, 1e-9):.2f} veces más rápida.")
