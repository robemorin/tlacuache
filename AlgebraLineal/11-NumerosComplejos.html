Ejemplo de Operaciones y Comprobación en Python:

El siguiente código de Python muestra cómo realizar las operaciones básicas de forma exacta usando SymPy y cómo realizar las multiplicaciones de comprobación del cociente obtenido:

import sympy as sp # Definir los números complejos simbólicamente z1 = 3 + 4*sp.I z2 = 1 - 2*sp.I print("--- Operaciones Aritméticas ---") suma = z1 + z2 resta = z1 - z2 mult = z1 * z2 division = sp.simplify(z1 / z2) print(f"z1 + z2 = {suma}") print(f"z1 - z2 = {resta}") print(f"z1 * z2 = {mult}") print(f"z1 / z2 = {division}\n") print("--- Comprobación de Resultados ---") # Comprobar la suma: s - z2 == z1 print(f"Comprobación Suma (s - z2): {suma - z2} (¿Es igual a z1?: {suma - z2 == z1})") # Comprobar la resta: d + z2 == z1 print(f"Comprobación Resta (d + z2): {resta + z2} (¿Es igual a z1?: {resta + z2 == z1})") # Comprobar la división: q * z2 == z1 comp_div = sp.simplify(division * z2) print(f"Comprobación División (q * z2): {comp_div} (¿Es igual a z1?: {comp_div == z1})")
